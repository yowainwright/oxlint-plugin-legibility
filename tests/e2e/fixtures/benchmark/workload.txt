export function select(first, second, third, fourth, fifth) {
  const value = first ? second ? third : fourth : fifth;
  return value;
}

export function flatten(items) {
  return items.map((item) => item.values).flat();
}

export function save(failed, value) {
  if (failed) {
    return null;
  } else {
    return value;
  }
}

export async function readValue(value) {
  return value;
}

export function enabled(kind) {
  return kind === "first" || kind === "second" || kind === "third" || kind === "fourth";
}
